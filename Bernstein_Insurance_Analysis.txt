###############################################################
## 3.1 Lognormal
###############################################################

d_lnorm <- function(x, par, log = FALSE) {

  dlnorm(
    x,
    meanlog = unname(par["meanlog"]),
    sdlog   = unname(par["sdlog"]),
    log     = log
  )
}


###############################################################
## 3.2 Gamma
## Parameterization: shape, rate
###############################################################

d_gamma <- function(x, par, log = FALSE) {

  dgamma(
    x,
    shape = unname(par["shape"]),
    rate  = unname(par["rate"]),
    log   = log
  )
}


###############################################################
## 3.3 Weibull
###############################################################

d_weibull <- function(x, par, log = FALSE) {

  dweibull(
    x,
    shape = unname(par["shape"]),
    scale = unname(par["scale"]),
    log   = log
  )
}


###############################################################
## 3.4 Burr XII
##
## F(x) =
## 1 - [1 + (x/scale)^c]^(-k)
###############################################################

d_burr12 <- function(x, par, log = FALSE) {

  cpar  <- unname(par["c"])
  kpar  <- unname(par["k"])
  scale <- unname(par["scale"])

  logdens <-
    log(cpar) +
    log(kpar) -
    log(scale) +
    (cpar - 1) * log(x / scale) -
    (kpar + 1) *
    log1p((x / scale)^cpar)

  if (log) {
    return(logdens)
  }

  exp(logdens)
}


###############################################################
## 3.5 Generalized Gamma
##
## flexsurv parameterization:
## mu, sigma, Q
###############################################################

d_ggamma <- function(x, par, log = FALSE) {

  flexsurv::dgengamma(
    x,
    mu    = unname(par["mu"]),
    sigma = unname(par["sigma"]),
    Q     = unname(par["Q"]),
    log   = log
  )
}


###############################################################
## 3.6 GB2
##
## Our notation:
## a,b,p,q
##
## GB2 package:
## shape1 = a
## scale  = b
## shape2 = p
## shape3 = q
###############################################################

d_gb2 <- function(x, par, log = FALSE) {

  a <- unname(par["a"])
  b <- unname(par["b"])
  p <- unname(par["p"])
  q <- unname(par["q"])

  dens <- GB2::dgb2(
    x,
    shape1 = a,
    scale  = b,
    shape2 = p,
    shape3 = q
  )

  if (log) {
    return(log(dens))
  }

  dens
}


###############################################################
## SECTION 4
## Main maximum-likelihood fitting function
###############################################################

fit_positive_distribution <- function(x, dist) {

  x <- as.numeric(x)

  x <- x[
    is.finite(x) &
    x > 0
  ]

  n <- length(x)


  #############################################################
  ## Lognormal
  #############################################################

  if (dist == "Lognormal") {

    theta0 <- c(
      mean(log(x)),
      log(sd(log(x)))
    )

    objective <- function(theta) {

      theta <- unname(theta)

      meanlog <- theta[1]

      sdlog <-
        exp(theta[2])

      ll <- dlnorm(
        x,
        meanlog = meanlog,
        sdlog   = sdlog,
        log     = TRUE
      )

      if (any(!is.finite(ll))) {
        return(1e100)
      }

      -sum(ll)
    }

    fit <- optim(
      par = theta0,
      fn = objective,
      method = "BFGS",
      control = list(
        maxit = 5000
      )
    )

    par <- c(
      meanlog =
        unname(fit$par[1]),

      sdlog =
        exp(unname(fit$par[2]))
    )
  }


  #############################################################
  ## Gamma
  #############################################################

  if (dist == "Gamma") {

    mx <- mean(x)

    vx <- var(x)

    shape0 <-
      mx^2 / vx

    rate0 <-
      mx / vx

    theta0 <-
      log(
        c(
          shape0,
          rate0
        )
      )

    objective <- function(theta) {

      theta <- unname(theta)

      shape <-
        exp(theta[1])

      rate <-
        exp(theta[2])

      ll <- dgamma(
        x,
        shape = shape,
        rate  = rate,
        log   = TRUE
      )

      if (any(!is.finite(ll))) {
        return(1e100)
      }

      -sum(ll)
    }

    fit <- optim(
      par = theta0,
      fn = objective,
      method = "BFGS",
      control = list(
        maxit = 5000
      )
    )

    par <- c(
      shape =
        exp(unname(fit$par[1])),

      rate =
        exp(unname(fit$par[2]))
    )
  }


  #############################################################
  ## Weibull
  #############################################################

  if (dist == "Weibull") {

    theta0 <- log(
      c(
        1,
        median(x)
      )
    )

    objective <- function(theta) {

      theta <- unname(theta)

      shape <-
        exp(theta[1])

      scale <-
        exp(theta[2])

      ll <- suppressWarnings(
        dweibull(
          x,
          shape = shape,
          scale = scale,
          log   = TRUE
        )
      )

      if (any(!is.finite(ll))) {
        return(1e100)
      }

      -sum(ll)
    }

    fit <- optim(
      par = theta0,
      fn = objective,
      method = "BFGS",
      control = list(
        maxit = 5000
      )
    )

    par <- c(
      shape =
        exp(unname(fit$par[1])),

      scale =
        exp(unname(fit$par[2]))
    )
  }


  #############################################################
  ## Burr XII
  #############################################################

  if (dist == "Burr12") {

    theta0 <- log(
      c(
        2,
        2,
        median(x)
      )
    )

    objective <- function(theta) {

      theta <- unname(theta)

      cpar <-
        exp(theta[1])

      kpar <-
        exp(theta[2])

      scale <-
        exp(theta[3])

      z <-
        log(x / scale)

      zz <-
        cpar * z

      ## Stable evaluation of log(1 + exp(zz))
      log1pexp <-
        ifelse(
          zz > 0,
          zz + log1p(exp(-zz)),
          log1p(exp(zz))
        )

      ll <-
        log(cpar) +
        log(kpar) -
        log(scale) +
        (cpar - 1) * z -
        (kpar + 1) * log1pexp

      if (any(!is.finite(ll))) {
        return(1e100)
      }

      -sum(ll)
    }

    fit <- optim(
      par = theta0,
      fn = objective,
      method = "BFGS",
      control = list(
        maxit = 10000,
        reltol = 1e-10
      )
    )

    par <- c(
      c =
        exp(unname(fit$par[1])),

      k =
        exp(unname(fit$par[2])),

      scale =
        exp(unname(fit$par[3]))
    )
  }


  #############################################################
  ## Generalized Gamma
  #############################################################

  if (dist == "GGamma") {

    theta0 <- c(
      mean(log(x)),
      log(sd(log(x))),
      0.5
    )

    objective <- function(theta) {

      theta <- unname(theta)

      mu <-
        theta[1]

      sigma <-
        exp(theta[2])

      Q <-
        theta[3]

      ll <- flexsurv::dgengamma(
        x,
        mu    = mu,
        sigma = sigma,
        Q     = Q,
        log   = TRUE
      )

      if (any(!is.finite(ll))) {
        return(1e100)
      }

      -sum(ll)
    }

    fit <- optim(
      par = theta0,
      fn = objective,
      method = "BFGS",
      control = list(
        maxit = 10000,
        reltol = 1e-10
      )
    )

    par <- c(
      mu =
        unname(fit$par[1]),

      sigma =
        exp(unname(fit$par[2])),

      Q =
        unname(fit$par[3])
    )
  }


  #############################################################
  ## GB2
  #############################################################

  if (dist == "GB2") {

    theta0 <- log(
      c(
        2,
        median(x),
        1,
        1
      )
    )

    log1pexp <- function(z) {

      ifelse(
        z > 0,
        z + log1p(exp(-z)),
        log1p(exp(z))
      )
    }

    objective <- function(theta) {

      theta <-
        unname(theta)

      a <-
        exp(theta[1])

      b <-
        exp(theta[2])

      p <-
        exp(theta[3])

      q <-
        exp(theta[4])

      logxb <-
        log(x) - log(b)

      z <-
        a * logxb

      ll <-
        log(a) -
        log(b) -
        lbeta(p, q) +
        (a * p - 1) * logxb -
        (p + q) * log1pexp(z)

      if (any(!is.finite(ll))) {
        return(1e100)
      }

      -sum(ll)
    }

    fit <- optim(
      par = theta0,
      fn = objective,
      method = "BFGS",
      control = list(
        maxit = 20000,
        reltol = 1e-10
      )
    )

    par <- c(
      a =
        exp(unname(fit$par[1])),

      b =
        exp(unname(fit$par[2])),

      p =
        exp(unname(fit$par[3])),

      q =
        exp(unname(fit$par[4]))
    )
  }


  #############################################################
  ## Information criteria
  #############################################################

  logLik_value <-
    -fit$value

  kpar <-
    length(par)

  AIC_value <-
    -2 * logLik_value +
    2 * kpar

  BIC_value <-
    -2 * logLik_value +
    log(n) * kpar


  #############################################################
  ## Return result
  #############################################################

  list(

    distribution = dist,

    par = par,

    logLik =
      logLik_value,

    AIC =
      AIC_value,

    BIC =
      BIC_value,

    convergence =
      fit$convergence,

    counts =
      fit$counts,

    message =
      fit$message

  )
}


###############################################################
## SECTION 5
## Fit all six distributions
###############################################################

fit_all_distributions <- function(x) {

  distributions <- c(
    "Lognormal",
    "Gamma",
    "Weibull",
    "Burr12",
    "GGamma",
    "GB2"
  )

  fits <- lapply(
    distributions,
    function(d) {

      fit_positive_distribution(
        x,
        d
      )

    }
  )

  names(fits) <-
    distributions

  summary_table <-
    do.call(
      rbind,
      lapply(
        fits,
        function(z) {

          data.frame(

            Distribution =
              z$distribution,

            LogLik =
              z$logLik,

            AIC =
              z$AIC,

            BIC =
              z$BIC,

            Convergence =
              z$convergence
          )
        }
      )
    )

  summary_table$AIC_rank <-
    rank(
      summary_table$AIC
    )

  summary_table$BIC_rank <-
    rank(
      summary_table$BIC
    )

  summary_table <-
    summary_table[
      order(summary_table$AIC),
    ]

  rownames(summary_table) <-
    NULL

  list(
    fits = fits,
    summary = summary_table
  )
}

###############################################################
## SECTION 8
## CDF functions for AD test
###############################################################

cdf_fit <- function(x, fit) {

  par <-
    fit$par

  dist <-
    fit$distribution


  if (dist == "Lognormal") {

    return(
      plnorm(
        x,
        meanlog =
          unname(par["meanlog"]),
        sdlog =
          unname(par["sdlog"])
      )
    )
  }


  if (dist == "Gamma") {

    return(
      pgamma(
        x,
        shape =
          unname(par["shape"]),
        rate =
          unname(par["rate"])
      )
    )
  }


  if (dist == "Weibull") {

    return(
      pweibull(
        x,
        shape =
          unname(par["shape"]),
        scale =
          unname(par["scale"])
      )
    )
  }


  if (dist == "Burr12") {

    cpar <-
      unname(par["c"])

    kpar <-
      unname(par["k"])

    scale <-
      unname(par["scale"])

    return(
      1 -
      (
        1 +
        (x / scale)^cpar
      )^(-kpar)
    )
  }


  if (dist == "GGamma") {

    return(
      flexsurv::pgengamma(
        x,
        mu =
          unname(par["mu"]),
        sigma =
          unname(par["sigma"]),
        Q =
          unname(par["Q"])
      )
    )
  }


  if (dist == "GB2") {

    return(
      GB2::pgb2(
        x,
        shape1 =
          unname(par["a"]),
        scale =
          unname(par["b"]),
        shape2 =
          unname(par["p"]),
        shape3 =
          unname(par["q"])
      )
    )
  }

}


###############################################################
## SECTION 9
## Quantile functions for bootstrap
###############################################################

quantile_fit <- function(u, fit) {

  par <-
    fit$par

  dist <-
    fit$distribution


  if (dist == "Lognormal") {

    return(
      qlnorm(
        u,
        meanlog =
          unname(par["meanlog"]),
        sdlog =
          unname(par["sdlog"])
      )
    )
  }


  if (dist == "Gamma") {

    return(
      qgamma(
        u,
        shape =
          unname(par["shape"]),
        rate =
          unname(par["rate"])
      )
    )
  }


  if (dist == "Weibull") {

    return(
      qweibull(
        u,
        shape =
          unname(par["shape"]),
        scale =
          unname(par["scale"])
      )
    )
  }


  if (dist == "Burr12") {

    cpar <-
      unname(par["c"])

    kpar <-
      unname(par["k"])

    scale <-
      unname(par["scale"])

    return(
      scale *
      (
        (1 - u)^(-1 / kpar) -
        1
      )^(1 / cpar)
    )
  }


  if (dist == "GGamma") {

    return(
      flexsurv::qgengamma(
        u,
        mu =
          unname(par["mu"]),
        sigma =
          unname(par["sigma"]),
        Q =
          unname(par["Q"])
      )
    )
  }


  if (dist == "GB2") {

    return(
      GB2::qgb2(
        u,
        shape1 =
          unname(par["a"]),
        scale =
          unname(par["b"]),
        shape2 =
          unname(par["p"]),
        shape3 =
          unname(par["q"])
      )
    )
  }

}


###############################################################
## SECTION 10
## Anderson-Darling statistic
###############################################################

AD_statistic <- function(x, fit) {

  x <-
    sort(
      as.numeric(x)
    )

  n <-
    length(x)

  Fval <-
    cdf_fit(
      x,
      fit
    )

  eps <-
    1e-12

  Fval <-
    pmin(
      pmax(
        Fval,
        eps
      ),
      1 - eps
    )

  i <-
    1:n

  AD <-
    -n -
    mean(
      (2 * i - 1) *
      (
        log(Fval) +
        log(
          1 -
          rev(Fval)
        )
      )
    )

  AD
}


###############################################################
## SECTION 11
## Parametric bootstrap AD test
###############################################################

AD_bootstrap <- function(
  x,
  dist,
  B = 100,
  seed = 12345
) {

  set.seed(seed)

  x <-
    as.numeric(x)

  x <-
    x[
      is.finite(x) &
      x > 0
    ]

  fit0 <-
    fit_positive_distribution(
      x,
      dist
    )

  AD_obs <-
    AD_statistic(
      x,
      fit0
    )

  AD_boot <-
    numeric(B)

  success <-
    0

  attempts <-
    0

  max_attempts <-
    5 * B


  while (
    success < B &&
    attempts < max_attempts
  ) {

    attempts <-
      attempts + 1


    u <-
      runif(
        length(x),
        min = 1e-8,
        max = 1 - 1e-8
      )


    xb <-
      try(
        quantile_fit(
          u,
          fit0
        ),
        silent = TRUE
      )


    if (
      inherits(
        xb,
        "try-error"
      ) ||
      any(!is.finite(xb)) ||
      any(xb <= 0)
    ) {
      next
    }


    fitb <-
      try(
        fit_positive_distribution(
          xb,
          dist
        ),
        silent = TRUE
      )


    if (
      inherits(
        fitb,
        "try-error"
      )
    ) {
      next
    }


    if (
      fitb$convergence != 0 ||
      !is.finite(fitb$logLik) ||
      fitb$logLik <= -1e90
    ) {
      next
    }


    adb <-
      try(
        AD_statistic(
          xb,
          fitb
        ),
        silent = TRUE
      )


    if (
      inherits(
        adb,
        "try-error"
      ) ||
      !is.finite(adb)
    ) {
      next
    }


    success <-
      success + 1

    AD_boot[success] <-
      adb
  }


  if (success == 0) {

    stop(
      "No successful bootstrap samples."
    )

  }


  AD_boot <-
    AD_boot[
      1:success
    ]


  pvalue <-
    (
      1 +
      sum(
        AD_boot >= AD_obs
      )
    ) /
    (
      success + 1
    )


  list(

    Distribution =
      dist,

    AD =
      AD_obs,

    p_value =
      pvalue,

    Successful_B =
      success,

    Failed =
      attempts - success,

    fit =
      fit0

  )
}


###############################################################
## SECTION 12
## Run AD bootstrap for all distributions
###############################################################

AD_all <- function(
  x,
  B = 100,
  seed = 1000
) {

  dists <- c(
    "Lognormal",
    "Gamma",
    "Weibull",
    "Burr12",
    "GGamma",
    "GB2"
  )

  res <-
    vector(
      "list",
      length(dists)
    )

  names(res) <-
    dists


  for (
    j in seq_along(dists)
  ) {

    cat(
      "\nRunning:",
      dists[j],
      "\n"
    )

    res[[j]] <-
      AD_bootstrap(
        x,
        dist = dists[j],
        B = B,
        seed = seed + j
      )
  }


  tab <-
    do.call(
      rbind,
      lapply(
        res,
        function(z) {

          data.frame(

            Distribution =
              z$Distribution,

            AD =
              z$AD,

            AD_pvalue =
              z$p_value,

            Successful_B =
              z$Successful_B,

            Failed =
              z$Failed
          )
        }
      )
    )


  rownames(tab) <-
    NULL


  list(
    results = res,
    summary = tab
  )
}

## 19.2 Bernstein component density
##
## For m = 2:
##
## U_i | N_i = nu_i
## ~ Beta(nu_i + 1, m - nu_i)
###############################################################

component_density_m2 <-
  function(
    u,
    state
  ) {

    d1 <-
      dbeta(
        u[1],
        shape1 =
          state[1] + 1,
        shape2 =
          2 - state[1]
      )

    d2 <-
      dbeta(
        u[2],
        shape1 =
          state[2] + 1,
        shape2 =
          2 - state[2]
      )

    d3 <-
      dbeta(
        u[3],
        shape1 =
          state[3] + 1,
        shape2 =
          2 - state[3]
      )

    d1 * d2 * d3
  }

###############################################################
## Gradient of negative log-likelihood for m = 3
###############################################################

grad_negloglik_theta3 <- function(theta) {

  gamma <- theta_to_gamma(theta)

  dens <- as.vector(D3 %*% gamma)

  if (any(dens <= 0) ||
      any(!is.finite(dens))) {

    return(rep(NA_real_, length(theta)))
  }

  ## Gradient with respect to gamma
  grad_gamma <-
    -as.vector(
      crossprod(
        D3,
        1 / dens
      )
    )

  ## Chain rule:
  ## gamma = gamma0 + Z theta
  grad_theta <-
    as.vector(
      crossprod(
        Z,
        grad_gamma
      )
    )

  grad_theta
}

###############################################################
## SECTION 33
## General Bernstein copula fitting function
## For m = 2 or m = 3
###############################################################

fit_Bernstein_general <- function(Utrain, m) {

  Utrain <- as.matrix(Utrain)

  n <- nrow(Utrain)

  states <- expand.grid(
    N1 = 0:(m - 1),
    N2 = 0:(m - 1),
    N3 = 0:(m - 1)
  )

  K <- nrow(states)


  #############################################################
  ## Component-density matrix
  #############################################################

  D <- matrix(
    0,
    nrow = n,
    ncol = K
  )

  for (r in 1:K) {

    nu <- as.numeric(states[r, ])

    D[, r] <-
      dbeta(
        Utrain[, 1],
        shape1 = nu[1] + 1,
        shape2 = m - nu[1]
      ) *
      dbeta(
        Utrain[, 2],
        shape1 = nu[2] + 1,
        shape2 = m - nu[2]
      ) *
      dbeta(
        Utrain[, 3],
        shape1 = nu[3] + 1,
        shape2 = m - nu[3]
      )
  }


  #############################################################
  ## Equality constraints
  ##
  ## sum gamma = 1
  ## and first m-1 marginal probabilities for each dimension
  #############################################################

  nr <- 1 + 3 * (m - 1)

  Aeq <- matrix(
    0,
    nrow = nr,
    ncol = K
  )

  beq <- c(
    1,
    rep(1 / m, 3 * (m - 1))
  )

  Aeq[1, ] <- 1

  row <- 2

  for (j in 1:3) {

    Nj <- states[[j]]

    for (k in 0:(m - 2)) {

      Aeq[row, ] <-
        as.numeric(Nj == k)

      row <- row + 1
    }
  }


  #############################################################
  ## Full SVD and null space
  #############################################################

  sv <- svd(
    Aeq,
    nu = 0,
    nv = ncol(Aeq)
  )

  tol <-
    max(dim(Aeq)) *
    max(sv$d) *
    .Machine$double.eps

  rankA <-
    sum(sv$d > tol)

  Z <-
    sv$v[
      ,
      (rankA + 1):ncol(Aeq),
      drop = FALSE
    ]


  #############################################################
  ## Independence starting point
  #############################################################

  gamma0 <-
    rep(
      1 / K,
      K
    )


  #############################################################
  ## theta -> gamma
  #############################################################

  theta_to_gamma_local <- function(theta) {

    as.vector(
      gamma0 +
      Z %*% theta
    )
  }


  #############################################################
  ## Negative log-likelihood
  #############################################################

  negloglik <- function(theta) {

    gamma <-
      theta_to_gamma_local(theta)

    if (any(gamma < 0)) {
      return(1e100)
    }

    dens <-
      as.vector(
        D %*% gamma
      )

    if (
      any(dens <= 0) ||
      any(!is.finite(dens))
    ) {
      return(1e100)
    }

    -sum(log(dens))
  }


  #############################################################
  ## Analytical gradient
  #############################################################

  grad_negloglik <- function(theta) {

    gamma <-
      theta_to_gamma_local(theta)

    dens <-
      as.vector(
        D %*% gamma
      )

    if (
      any(dens <= 0) ||
      any(!is.finite(dens))
    ) {
      return(
        rep(
          NA_real_,
          length(theta)
        )
      )
    }

    grad_gamma <-
      -as.vector(
        crossprod(
          D,
          1 / dens
        )
      )

    as.vector(
      crossprod(
        Z,
        grad_gamma
      )
    )
  }


  #############################################################
  ## Optimization
  #############################################################

  theta0 <-
    rep(
      0,
      ncol(Z)
    )

  fit <- constrOptim(
    theta = theta0,
    f = negloglik,
    grad = grad_negloglik,
    ui = Z,
    ci = -gamma0,
    method = "BFGS",
    control = list(
      maxit = 20000,
      reltol = 1e-9
    ),
    outer.iterations = 100,
    outer.eps = 1e-7
  )


  gamma_hat <-
    theta_to_gamma_local(
      fit$par
    )


  list(
    gamma = gamma_hat,
    states = states,
    m = m,
    logLik = -negloglik(fit$par),
    convergence = fit$convergence
  )
}

###############################################################
## SECTION 34
## Bernstein copula density at one observation
###############################################################

dBernstein_point <- function(
  u,
  gamma,
  states,
  m
) {

  K <- nrow(states)

  component <- numeric(K)

  for (r in 1:K) {

    nu <- as.numeric(
      states[r, ]
    )

    component[r] <-
      dbeta(
        u[1],
        shape1 = nu[1] + 1,
        shape2 = m - nu[1]
      ) *
      dbeta(
        u[2],
        shape1 = nu[2] + 1,
        shape2 = m - nu[2]
      ) *
      dbeta(
        u[3],
        shape1 = nu[3] + 1,
        shape2 = m - nu[3]
      )
  }

  sum(
    gamma * component
  )
}

###############################################################
## SECTION 35
## Leave-One-Out Cross-Validation
###############################################################

LOOCV_Bernstein <- function(Umat, m) {

  Umat <- as.matrix(Umat)

  n <- nrow(Umat)

  logscore <- rep(
    NA_real_,
    n
  )

  convergence <- rep(
    NA_integer_,
    n
  )


  for (i in 1:n) {

    cat(
      "m =", m,
      "  left-out observation =", i,
      "of", n, "\n"
    )


    Utrain <-
      Umat[-i, , drop = FALSE]

    utest <-
      Umat[i, ]


    fit_i <-
      try(
        fit_Bernstein_general(
          Utrain,
          m
        ),
        silent = TRUE
      )


    if (
      inherits(
        fit_i,
        "try-error"
      )
    ) {

      cat(
        "   Fit failed.\n"
      )

      next
    }


    convergence[i] <-
      fit_i$convergence


    dens_i <-
      dBernstein_point(
        u = utest,
        gamma = fit_i$gamma,
        states = fit_i$states,
        m = m
      )


    if (
      is.finite(dens_i) &&
      dens_i > 0
    ) {

      logscore[i] <-
        log(dens_i)

    }

  }


  list(

    m = m,

    logscore =
      logscore,

    total_logscore =
      sum(
        logscore,
        na.rm = TRUE
      ),

    mean_logscore =
      mean(
        logscore,
        na.rm = TRUE
      ),

    successful =
      sum(
        is.finite(logscore)
      ),

    convergence =
      convergence
  )
}

## SECTION 38
## FAST simulation from fitted Bernstein copula, m = 2
###############################################################

rBernstein_m2_fast <- function(B, gamma_hat, states) {

  ## Draw mixture component
  comp <- sample.int(
    n = nrow(states),
    size = B,
    replace = TRUE,
    prob = gamma_hat
  )

  N1 <- states$N1[comp]
  N2 <- states$N2[comp]
  N3 <- states$N3[comp]

  ## Conditional Beta draws
  U1_sim <- rbeta(
    B,
    shape1 = N1 + 1,
    shape2 = 2 - N1
  )

  U2_sim <- rbeta(
    B,
    shape1 = N2 + 1,
    shape2 = 2 - N2
  )

  U3_sim <- rbeta(
    B,
    shape1 = N3 + 1,
    shape2 = 2 - N3
  )

  data.frame(
    U1 = U1_sim,
    U2 = U2_sim,
    U3 = U3_sim
  )
}

risk_measures <- function(S, kappa = 0.95) {

  VaR <- unname(
    quantile(
      S,
      probs = kappa,
      type = 8
    )
  )

  tail_index <- S > VaR

  TVaR <- mean(
    S[tail_index]
  )

  c(
    VaR  = VaR,
    TVaR = TVaR
  )
}

## TVaR capital allocation
###############################################################

TVaR_allocation <- function(
  X1,
  X2,
  X3,
  kappa = 0.95
) {

  S <- X1 + X2 + X3

  VaR <- unname(
    quantile(
      S,
      probs = kappa,
      type = 8
    )
  )

  tail_index <- S > VaR

  TVaR_S <- mean(
    S[tail_index]
  )

  TVaR_X1 <- mean(
    X1[tail_index]
  )

  TVaR_X2 <- mean(
    X2[tail_index]
  )

  TVaR_X3 <- mean(
    X3[tail_index]
  )

  c(
    VaR_S   = VaR,
    TVaR_S  = TVaR_S,
    TVaR_X1 = TVaR_X1,
    TVaR_X2 = TVaR_X2,
    TVaR_X3 = TVaR_X3,
    Sum_alloc =
      TVaR_X1 +
      TVaR_X2 +
      TVaR_X3
  )
}
